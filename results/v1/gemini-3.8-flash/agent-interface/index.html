--[[ NexusAgent 生成 · 高并发原子减库存脚本 KEYS[1]: 库存主 Key (例如: stock:sku_1001) ARGV[1]: 扣减数量 ]] local stock_key = KEYS[1] local deduct_num = tonumber(ARGV[1]) local current_stock = tonumber(redis.call(GET, stock_key)) -- 如果键不存在，返回 -1 错误码 if not current_stock then return -1 end -- 如果库存不足，返回 0 拒绝 if current_stock < deduct_num then return 0 end -- 执行原子扣减 redis.call(DECRBY, stock_key, deduct_num) return 1
